@echo off
color 0e
title Reparacion e Integridad del Sistema (SFC y DISM)

echo ===================================================
echo   COMPROBACION E INTEGRIDAD DEL SISTEMA
echo ===================================================
echo.

echo 1. Diagnosticando salud de la imagen de Windows (DISM)...
Dism /Online /Cleanup-Image /CheckHealth

echo.
echo 2. Reparando la imagen de Windows (DISM)...
Dism /Online /Cleanup-Image /RestoreHealth

echo.
echo 3. Escaneando y reparando archivos del sistema (SFC)...
sfc /scannow

echo.
echo ===================================================
echo   PROCESO DE REPARACION Y DIAGNOSTICO FINALIZADO
echo ===================================================
echo.
pause
