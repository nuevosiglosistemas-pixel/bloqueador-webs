@echo off
:: Configuración inicial del script
cd /d "%~dp0"
title Configurador y Bloqueador de Webs - Nuevos Siglo Sistemas

:: Mostrar mensaje visual flotante en pantalla (sin ventana DOS fea)
start "" mshta "javascript:var sh=new ActiveXObject('WScript.Shell'); sh.Popup('Activando políticas de seguridad local\n\nAres sistemas Nuevo Siglo', 2, 'Seguridad - Nuevo Siglo Sistemas', 64);"

echo ========================================================
echo   INICIANDO CONFIGURACION Y DESCARGA DE DIRECTIVAS
echo ========================================================

echo.
echo [1/3] Verificando y limpiando archivos anteriores...
if exist configuracion_inicial.txt (
    echo Borrando configuracion_inicial.txt anterior...
    del configuracion_inicial.txt
)
if exist configuracion_inicial.bat (
    echo Borrando configuracion_inicial.bat anterior...
    del configuracion_inicial.bat
)

echo.
echo [2/3] Descargando la nueva version desde GitHub...
powershell -Command "Invoke-WebRequest -Uri 'https://raw.githubusercontent.com/nuevosiglosistemas-pixel/bloqueador-webs/main/configuracion_inicial.txt' -OutFile 'configuracion_inicial.bat'"

echo.
echo [3/3] Ejecutando el archivo descargado...
if exist configuracion_inicial.bat (
    :: Ejecuta el archivo descargado ignorando cualquier "pause" interno
    call configuracion_inicial.bat <nul
    echo.
    echo ¡Proceso completado con exito!
) else (
    echo.
    echo [ERROR] No se pudo descargar o ejecutar el archivo de configuracion.
)

echo ========================================================
echo Proceso terminado.
timeout /t 2 /nobreak >nul
exit
