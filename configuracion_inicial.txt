@echo off
title Limpiador de Temporales y Papelera
color 0A

echo [1/3] Limpiando archivos temporales del sistema...
del /s /f /q C:\Windows\Temp\*.* 2>nul
rd /s /q C:\Windows\Temp 2>nul
md C:\Windows\Temp 2>nul

echo [2/3] Limpiando temporales del usuario (%USERNAME%)...
del /s /f /q "%USERPROFILE%\AppData\Local\Temp\*.*" 2>nul
rd /s /q "%USERPROFILE%\AppData\Local\Temp" 2>nul
md "%USERPROFILE%\AppData\Local\Temp" 2>nul

echo [3/3] Vaciando la papelera de reciclaje...
powershell -NoProfile -Command "Clear-RecycleBin -Force -ErrorAction SilentlyContinue"

echo.
echo ==========================================
echo ¡Limpieza completada con exito!
echo ==========================================
pause
