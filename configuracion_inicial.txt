@echo off
echo Limpiando archivos temporales y papelera de reciclaje...

:: Borrar temporales del usuario actual
del /s /f /q "%userprofile%\AppData\Local\Temp\*.*"
rd /s /q "%userprofile%\AppData\Local\Temp"
md "%userprofile%\AppData\Local\Temp"

:: Borrar temporales de Windows
del /s /f /q "C:\Windows\Temp\*.*"
rd /s /q "C:\Windows\Temp"
md "C:\Windows\Temp"

:: Vaciar la papelera de reciclaje en todas las unidades
powershell -Command "Clear-RecycleBin -Force -ErrorAction SilentlyContinue"

echo.
echo Limpieza completada con exito.
timeout /t 5
exit
