@echo off
:: Asegurar permisos de Administrador de forma silenciosa
>nul 2>&1 "%SYSTEMROOT%\system32\cacls.exe" "%SYSTEMROOT%\system32\config\system"
if '%errorlevel%' NEQ '0' (
    goto UACPrompt
) else ( goto gotAdmin )

:UACPrompt
echo Set UAC = CreateObject^("Shell.Application"^) > "%temp%\getadmin.vbs"
echo UAC.ShellExecute "%~s0", "", "", "runas", 1 >> "%temp%\getadmin.vbs"
"%temp%\getadmin.vbs"
exit /B

:gotAdmin
if exist "%temp%\getadmin.vbs" ( del "%temp%\getadmin.vbs" )
pushd "%CD%"
CD /D "%~dp0"

:: ----------------------------------------------------
:: CAMBIAR EL TÍTULO DE LA BARRA DE LA VENTANA
:: ----------------------------------------------------
title Aplicando directivas de optimizacion de inicio de sesion

:: Mensajes iniciales en pantalla
echo =====================================================
echo    APLICANDO DIRECTIVAS DE OPTIMIZACION
echo =====================================================
echo.

:: 1. Limpiando archivos temporales del usuario
echo [1/6] Limpiando temporales del usuario...
del /s /f /q "%temp%\*.*" >nul 2>&1
rd /s /q "%temp%" >nul 2>&1
md "%temp%" >nul 2>&1

:: 2. Limpiando archivos temporales del sistema
echo [2/6] Limpiando temporales del sistema...
del /s /f /q "C:\Windows\Temp\*.*" >nul 2>&1
rd /s /q "C:\Windows\Temp" >nul 2>&1
md "C:\Windows\Temp" >nul 2>&1

:: 3. Limpiando caché de actualizaciones de Windows
echo [3/6] Limpiando cache de Windows Update...
net stop wuauserv >nul 2>&1
del /s /f /q "C:\Windows\SoftwareDistribution\Download\*.*" >nul 2>&1
net start wuauserv >nul 2>&1

:: 4. Limpiando archivos Prefetch del sistema
echo [4/6] Limpiando archivos Prefetch...
del /s /f /q "C:\Windows\Prefetch\*.*" >nul 2>&1

:: 5. Limpiando caché de DNS
echo [5/6] Vaciando cache de DNS...
ipconfig /flushdns >nul 2>&1

:: 6. Vaciando la Papelera de reciclaje
echo [6/6] Vaciando la Papelera de reciclaje...
powershell -NoProfile -Command "Clear-RecycleBin -Force -ErrorAction SilentlyContinue" >nul 2>&1

echo.
echo =====================================================
echo    PROCESO COMPLETADO CON EXITO
echo =====================================================
timeout /t 3 >nul
exit
