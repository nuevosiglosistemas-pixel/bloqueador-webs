@echo off
color 0a
title Limpieza Tecnica y Optimizacion de Windows

echo ===================================================
echo   INICIANDO LIMPIEZA DE ARCHIVOS TEMPORALES
echo ===================================================
echo.

echo 1. Cerrando procesos no esenciales en segundo plano...
taskkill /f /im utorrent.exe 2>nul
echo.

echo 2. Limpiando carpeta Temp de Windows...
del /s /f /q "%systemroot%\Temp\*.*" 2>nul
for /d %%i in ("%systemroot%\Temp\*") do rmdir /s /q "%%i" 2>nul

echo 3. Limpiando carpeta Temp de Usuario...
del /s /f /q "%temp%\*.*" 2>nul
for /d %%i in ("%temp%\*") do rmdir /s /q "%%i" 2>nul

echo 4. Limpiando cache de Prefetch...
del /s /f /q "%systemroot%\Prefetch\*.*" 2>nul

echo 5. Limpiando cache de DNS...
ipconfig /flushdns >nul

echo.
echo ===================================================
echo   LIMPIEZA COMPLETADA CON EXITO
echo ===================================================
echo.
pause
