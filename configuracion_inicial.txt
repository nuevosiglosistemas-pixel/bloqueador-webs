@echo off
Title Optimizador Total de Discos
:: Solicita permisos de administrador automáticamente si no los tiene
>nul 2>&1 "%SYSTEMROOT%\system32\cacls.exe" "%SYSTEMROOT%\system32\config\system"
if '%errorlevel%' NEQ '0' (
    goto UACPrompt
) else ( goto gotAdmin )

:UACPrompt
echo Set UAC = CreateObject^("Shell.Application"^) > "%temp%\getadmin.vbs"
echo UAC.ShellExecute "%~s0", "", "", "runas", 1 >> "%temp%\getadmin.vbs"
"%temp%\getadmin.vbs"
exit /B

:gotAdmin
if exist "%temp%\getadmin.vbs" ( del "%temp%\getadmin.vbs" )
pushd "%CD%"
CD /D "%~dp0"

echo ========================================================
echo   Optimizando todas las unidades de forma automatica...
echo ========================================================
echo.

:: El comando /C optimiza todas las unidades posibles y /H ejecuta la tarea con prioridad alta
defrag /C /O /H

exit
