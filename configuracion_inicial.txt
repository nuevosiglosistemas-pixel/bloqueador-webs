@echo off
title Mantenimiento y Limpieza del Sistema
color 0A

echo ========================================================
echo Limpiando archivos temporales del usuario...
echo ========================================================
del /s /f /q "%userprofile%\AppData\Local\Temp\*.*" 2>nul
rd /s /q "%userprofile%\AppData\Local\Temp" 2>nul
md "%userprofile%\AppData\Local\Temp"

echo.
echo ========================================================
echo Limpiando archivos temporales del sistema (Windows\Temp)...
echo ========================================================
del /s /f /q "%windir%\Temp\*.*" 2>nul
rd /s /q "%windir%\Temp" 2>nul
md "%windir%\Temp"

echo.
echo ========================================================
echo Limpiando Prefetch...
echo ========================================================
del /s /f /q "%windir%\Prefetch\*.*" 2>nul

echo.
echo ========================================================
echo Vaciando la Papelera de Reciclaje...
echo ========================================================
powershell -Command "Clear-RecycleBin -Force -ErrorAction SilentlyContinue"

exit
