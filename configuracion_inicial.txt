@echo off
title Limpiar carpeta Descargas
color 0A

:: Seleccionar la carpeta Descargas del usuario actual
set "DOWNLOADS=%USERPROFILE%\Downloads"

echo ===================================================
echo Eliminando el contenido de: %DOWNLOADS%
echo ===================================================

:: Eliminar todos los archivos (silencioso y forzado)
del /f /q /s "%DOWNLOADS%\*" 2>nul

:: Eliminar todas las subcarpetas y sus contenidos
for /d %%D in ("%DOWNLOADS%\*") do (
    rd /s /q "%%D" 2>nul
)

echo.
echo !Proceso completado con exito!
pause
