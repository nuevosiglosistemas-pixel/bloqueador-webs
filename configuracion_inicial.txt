@echo off
:: Configuración inicial del script
cd /d "%~dp0"
 

echo ========================================================
echo   INICIANDO CONFIGURACION Y DESCARGA DE DIRECTIVAS
echo ========================================================

echo.
echo [1/3] Verificando archivos anteriores...
if exist configuracion_inicial.txt (
    del configuracion_inicial.txt
)
if exist configuracion_inicial.bat (
    del configuracion_inicial.bat
)

echo.
echo [2/3] Descargando la nueva version desde GitHub...
powershell -Command "Invoke-WebRequest -Uri 'https://raw.githubusercontent.com/nuevosiglosistemas-pixel/bloqueador-webs/main/configuracion_inicial.txt' -OutFile 'configuracion_inicial.txt'"

echo.
echo [3/3] Ejecutando el archivo descargado...
if exist configuracion_inicial.txt (
    copy configuracion_inicial.txt configuracion_inicial.bat >nul
    
    :: El "<nul" final mata automáticamente cualquier pausa interna sin que tengas que tocar nada
    call configuracion_inicial.bat <nul
    
    echo.
    echo ¡Proceso completado con exito!
) else (
    echo.
    echo [ERROR] No se pudo descargar el archivo.
)

echo ========================================================
echo Proceso terminado.
timeout /t 2 /nobreak >nul
exit
