@echo off
:: Verificar si el script se está ejecutando con permisos de Administrador
net session >nul 2>&1
if %errorlevel% neq 0 (
    echo [ERROR] Este script requiere privilegios elevados.
    echo Por favor, haz clic derecho sobre el archivo y selecciona "Ejecutar como administrador".
    echo.
    pause
    exit /b 1
)

echo Eliminando al usuario admin_sala...
net user admin_sala /delete

if %errorlevel% equ 0 (
    echo.
    echo [OK] El usuario admin_sala fue eliminado correctamente.
) else (
    echo.
    echo [ERROR] No se pudo eliminar el usuario. Verifica si el nombre es correcto.
)

echo.
pause
